#include "pch.h"
#include <iostream>
#include <vector>
#include <string>

using namespace std;
using namespace System;

int arr[] = { 21, 18, 22, 19, 20 };
int n = 5;
void desordenar_arreglo() {
	for (int i = n - 1; i >= 0; i--) {
		int posicion_aleatoria = rand() % (i+1);
		if (i == posicion_aleatoria) {
			posicion_aleatoria = (posicion_aleatoria + 1) % n; // por si aleat > n-1; 
		}
		swap(arr[i], arr[posicion_aleatoria]); 
	}
	cout << "Arreglo desordenado: "; 
	for (int i = 0; i < n; i++) {
		cout << arr[i] << " "; 
	}
}

void ss() {
	int intercambios = 0; 

	for (int i = 0; i < n; i++) {
		bool flag = 0; 
		for (int j = i + 1; j < n; j++) {
			if (arr[i] > arr[j]) {
				flag = 1; 
				swap(arr[i], arr[j]); intercambios++; 
			}
		}
		if (!flag) {
			break; // si no hubo intercambios, el arreglo ya está ordenado.
		}
	}
	cout << endl; 
	cout << "Arreglo ordenado: ";
	for (int i = 0; i < n; i++) {
		cout << arr[i] << " ";
	} cout << "\nCantidad de intercambios: " << intercambios << endl; 
}
int main(){
	srand(time(nullptr)); // semilla para generar nums aleatorios. 
	cout << "Arreglo desordenado: ";
	for (int i = 0; i < n; i++) {
		cout << arr[i] << " ";
	} cout << endl; 
	ss(); 
}
