//T = int 
template <class T>
class Nodo {
public:
	T dato;
	Nodo<T>* next;
	Nodo(T dat) { next = nullptr; dato = dat; }
};

template<class T>
class ListaEnlazada {
private:
	Nodo<T>* cabeza;
public:
	ListaEnlazada() { cabeza = nullptr; }

	void Anadir(T dato) {
		if (dato < 100) {
			cout << "No se ha podido anadir el dato. \n";
			return;
		}
		if (cabeza == nullptr) {
			cabeza = new Nodo<T>(dato);
			cout << "El dato ha sido anadido.\n";
			return;
		}
		Nodo<T>* aux = cabeza;
		while (aux->next != nullptr) {
			aux = aux->next;
		}
		aux->next = new Nodo<T>(dato);
		cout << "El dato ha sido anadido.\n";
	}

	void MostrarMultiplos() {
		cout << "Valores multiplos de 5: ";
		Nodo<T>* aux = cabeza;
		while (aux != nullptr) {
			if (aux->dato % 5 == 0) {
				cout << aux->dato << " ";
			}
			aux = aux->next;
		}
		cout << endl;
	}

	void EliminarLectura(int posicion) {
		if (cabeza == nullptr) {
			cout << "Lista vacia\n";
			return;
		}
		if (posicion % 2 == 0) {
			cout << "Posicion invalida para borrar. \n";
			return;
		}

		bool yes = 0; //se ha podido borrar un valor 
		int n = 0; //valores procesados

		if (posicion == 0) {
			Nodo<T>* temp = cabeza;
			cabeza = cabeza->next;
			delete temp;
			return;
		}

		Nodo<T>* actual = cabeza;
		for (int i = 0; i < posicion - 1; i++) { //recorrer hasta posicion - 1 
			if (actual == nullptr) break;
			actual = actual->next;
		} //opciones de retorno: FINAL || nullptr || PENULTIMO->FINAL || ANTEPENULTIMO -> POSICION-> FINAL

		if (actual == nullptr || actual->next == nullptr) {
			cout << "No existe esa posicion en la lista enlazada.\n";
			return;
		}

		Nodo<T>* temp = actual->next;
		// obviar el valor de el medio 
		actual->next = actual->next->next;
		delete temp;
	}

	void Modificar(int posicion, int valor) {
		if (valor < 100 || valor > 500)return; 
		int cnt = 0;
		Nodo<T>* aux = cabeza;
		for (int i = 0; i < posicion; i++) {
			if (aux == nullptr) break;
			aux = aux->next;
		}
		if (aux == nullptr) {
			cout << "Posicion no valida.\n";
			return;
		}
		aux->dato = valor;
		cout << "Posicion actualizada. \n";
	}

	void ImprimirTodo() {
		cout << "Lista actual: ";
		Nodo<T>* aux = cabeza;
		while (aux != nullptr) {
			cout << aux->dato << " ";
			aux = aux->next;
		}
		cout << "\n";
	}
};
