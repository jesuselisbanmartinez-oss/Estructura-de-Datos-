class Nodo {
public: 
    T dato; 
    Nodo* next; // conexión que tiene 
    Nodo(T datoo) {
        next = nullptr; 
        dato = datoo; 
    }
    void imprimirDato() {
        cout << dato;
    }
   // Nodo(T datoo): dato(datoo), next(nullptr){}
};
