int recursividad(int numero) {
	if (numero < 1) return 0; 
	int suma = 0; 
	suma += (numero % 10); // extraer el ultimo digito
	suma += recursividad(numero / 10); 
	//suma del numero sin considerar el ultimo digito
	return suma; 
}
