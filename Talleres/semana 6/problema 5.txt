#include "pch.h"
#include <iostream>
#include <vector>
#include <string>

using namespace std;
using namespace System;

struct Cliente {
	string nombre; 
	int dni; 
};

Cliente clientes[] = {
 {"Juan", 12345678}, // COMPARADO CON CARLOS 
 {"Ana", 87654321},
 {"ana", 47554499},
 {"Carlos", 83746873}, // COMPARADO CON JUAN 
 {"Luis", 98765432},
 {"Maria", 12345678},
 {"Pedro", 45678901} // 7  COMPARADO CON CARLOS, JUAN 
};

void shell(int n) {
	for (int gap = n / 2; gap > 0; gap /= 2) { // logaritmico n/2/2/2 
		for (int i = gap; i < n; i++) {
			Cliente temp = clientes[i];
			int j; 
			for (j = i; j >= gap && clientes[j - gap].dni > temp.dni; j -= gap) {
				clientes[j] = clientes[j - gap];
			}
			clientes[j] = temp; 
		}
	}
	cout << "Clientes ordenados por DNI:" << endl;
	for (int i = 0; i < n; i++) {
		cout << "DNI: " << clientes[i].dni << ", NOMBRE: " << clientes[i].nombre << endl;
	}
}

int main(){
	int  n= sizeof(clientes)/ sizeof(clientes[0]); // para los arreglos
	shell(n); 

	
}
