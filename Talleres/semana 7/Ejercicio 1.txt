// posicion 0: 4 [4, [recursividad]]  
// po 1: 3 -> recursividad(i+1) [2 1]
int** recursividad(int posi, int arreglo[]) { // [ 1 2 3 4]
    // caso base: 
    int** ar = new int* [n - posi];
    if (posi == n - 1) { // size es 1 
        ar[0] = new int(arreglo[n - posi - 1]); // arreglo de 0; 
        return ar;
    }
    int** aux = recursividad(posi + 1, arreglo);
    for (int i = 1; i < n - posi; i++) {
        ar[i] = aux[i - 1];
    }
    ar[0] = new int(arreglo[n - posi - 1]); delete[] aux;
    return ar;
}
