#include <bits/stdc++.h>
using namespace std;

class Estudiante{
    public: 
    string DNI; 
    string nombre; 
    int puntaje_simulacro; 

    Estudiante(string d, string n, int p): DNI(d), nombre(n), puntaje_simulacro(p){}
}; 

int partition(vector<Estudiante>&arreglo, int l, int h){
    Estudiante pivote= arreglo[h]; //pivote = *(ind=h)
    int p= l; 
    for(int i=l; i<= h-1; i++){
        if(arreglo[i].puntaje_simulacro > pivote.puntaje_simulacro){
            Estudiante aux= arreglo[i]; 
            arreglo[i] = arreglo[p]; 
            arreglo[p] = aux;  p++;   
        }
    } 
    Estudiante aux= arreglo[p]; 
    arreglo[p] = arreglo[h];
    arreglo[h] = aux; 

    return p;  // retorna el ind ordenado descendentemenete. 
}


Estudiante quick_select(vector<Estudiante>&lista, int l, int h, int k){
    if(l == h)return lista[l]; //solo hay un valor 
    int p= partition(lista, l, h); 
    int Em= p - l + 1; //estamos indexando las posiciones en 0 
    // k = 6 
    // pivote = 3 -> 
    // si vamos a buscar por el lado de la derecha, buscaremos ahora el 3er valor (no 6)
    if(k == Em)return lista[p]; 

    else if(Em > k){ // tenemos que buscar por el lado izq
      return quick_select(lista, l, p-1, k); 
    }
    else{
        return quick_select(lista, p+1, h, k - Em); 
    }
}


int main() {
	vector<Estudiante> lista = {

        Estudiante("71001122",	"Sara Diaz",	1680)
    }; 
    Estudiante elegido = quick_select(lista, 0, int(lista.size())-1, 3); 
    cout << "El 2do estudiante con mayor calificacion es: " << elegido.nombre << "\n";
