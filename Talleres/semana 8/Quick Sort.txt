#include <bits/stdc++.h>
using namespace std;
int arreglo[100000]; // 1e5 elementos 
int cnt=0; 

// QUICK SORT 

int partition(int l, int h){
    int pivote= arreglo[h]; //pivote = *(ind=h)
    int p= l-1; 
    for(int i=l; i<= h-1; i++){
        if(arreglo[i] < pivote){
            p++; // existe un elemento menor a pivote 
            //swap 
            int aux= arreglo[i]; 
            arreglo[i] = arreglo[p]; 
            arreglo[p] = aux;    
        }
    } 

    int aux= arreglo[p +1]; 
    arreglo[p+1] = arreglo[h]; //donde se encuentra pivote
    arreglo[h] = aux; 

    return p+1; // pivoste es el p valor mas peqeño 
}

void quick_sort(int l, int h){ // divide y venceras 
    if(l >= h) return; 
    int p= partition(l, h); 
    quick_sort(l, p-1); 
    quick_sort(p+1, h); 
    return; 
}

int main() {
    srand(time(nullptr)); 
    int n= 1e5-1; 
    for(int i=0; i < n; i++){ //creamos el arreglo desordenado 
        arreglo[i] = rand()%100000;    
    }
    cout << "Arreglo desordenado: (Primeros 10 elementos)\n"; 
    for(int i=0; i < 10; i++){ //creamos el arreglo desordenado 
       cout << arreglo[i] << " "; 
    }

    //merge_sort(0, n-1); 
    quick_sort(0, n-1); 
    cout << "\nArreglo ordenado: (Primeros 10 elementos)\n"; 
    for(int i=0; i < 10; i++){ 
       cout << arreglo[i] << " "; 
    }
    cout << "\nCantidad de iteraciones: " << cnt << "\n";
 //1e10 a 1e6 iteraciones
}
